async function carregarCardapio() { try { const res = await fetch('/api/espetinhos'); const itens = await res.json(); const container = document.getElementById('cardapio'); container.innerHTML = ''; if (itens.length === 0) { container.innerHTML = '

Nenhum espetinho cadastrado.

'; return; } itens.forEach(item => { container.innerHTML += `

${item.nome}

${item.descricao}

R$ ${parseFloat(item.preco).toFixed(2).replace('.', ',')}

`; }); } catch (err) { document.getElementById('cardapio').innerHTML = '

Erro ao carregar dados da API.

'; } } carregarCardapio();
